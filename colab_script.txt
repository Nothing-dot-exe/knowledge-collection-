def extract_repo_name(url: str) -> str:
    cleaned = url.rstrip("/").removesuffix(".git")
    parts = cleaned.split("/")
    return f"{parts[-2]}/{parts[-1]}" if len(parts) >= 2 else cleaned

            merged_papers = list(local_reg.get("papers", []))

            merged_titles = list(local_reg.get("titles", []))
            for p, h, t in zip(remote_reg.get("papers", []), remote_reg.get("hashes", []), remote_reg.get("titles", [])):
                if h not in merged_hashes:
                    merged_papers.append(p)
                    merged_hashes.append(h)
                    merged_titles.append(t)
            clean_p, clean_h, clean_t = [], [], []
            for p, h, t in zip(merged_papers, merged_hashes, merged_titles):
                if t.strip() in ("<!-- image -->", "!-- image", "Document") and p.startswith("tg_"):

def compute_sha256(filepath: Path) -> str:
    h = hashlib.sha256()
    with open(filepath, "rb") as f:
        while chunk := f.read(131072):
            h.update(chunk)
    return h.hexdigest()

def check_deduplication(file_hash: str | None, title: str, reg: dict[str, list[str]]) -> tuple[bool, str, str]:
    hashes = reg.get("hashes", [])
    titles = reg.get("titles", [])
    if file_hash and file_hash in hashes:
        idx = hashes.index(file_hash)
        return True, f"KB-{idx+1:04d}", f"Binary SHA-256 match with {titles[idx]}"
    clean_search = re.sub(r"[^\w\s]", "", title).strip().lower()
    for idx, t in enumerate(titles):
        clean_existing = re.sub(r"[^\w\s]", "", t).strip().lower()
        if clean_search and clean_search == clean_existing:
            return True, f"KB-{idx+1:04d}", f"Exact title match with '{t}'"
        if len(clean_search) > 15 and len(clean_existing) > 15:
            ratio = fuzz.ratio(clean_search, clean_existing)
            token_ratio = getattr(fuzz, "token_set_ratio", fuzz.ratio)(clean_search, clean_existing)
            if ratio >= 85 or token_ratio >= 88:
                return True, f"KB-{idx+1:04d}", f"Fuzzy title match ({max(ratio, token_ratio)}%) with '{t}'"
    return False, "", ""

def sanitize_markdown_zero_links(md: str) -> str:
    md = re.sub(r'\[([^\]]+)\]\(https?://[^\)]+\)', r'\1', md)
    md = re.sub(r'https?://\S+', '', md)
    md = re.sub(r'!\[[^\]]*\]\([^\)]*\)', '', md)
    md = re.sub(r'\[\s*\]', '', md)
    return re.sub(r'\n{3,}', '\n\n', md).strip()

def extract_pdf(pdf_path: Path) -> tuple[str, int]:
    conv = get_docling_converter()
    res = conv.convert(str(pdf_path))
    raw_md = res.document.export_to_markdown()
    pages = 1
    try:
        pages = len(getattr(res.document, "pages", [1]))
    except Exception:
        pass
    return sanitize_markdown_zero_links(raw_md), max(1, pages)

def chunk_text(text: str, kb_id: str, title: str) -> list[dict]:
    words = text.split()
    records = []
    start = 0
    idx = 1
    total = len(words)
    while start < total:
        end = min(start + 2000, total)
        c_str = " ".join(words[start:end])
        records.append({
            "messages": [
                {"role": "system", "content": f"You are an expert AI researcher analyzing document {kb_id}: '{title}'."},
                {"role": "user", "content": f"Provide comprehensive analysis of the following section from '{title}':\n\n{c_str}"},
                {"role": "assistant", "content": f"Here is the detailed synthesis of section {idx} from '{title}':\n\n{c_str}"}
            ],
            "metadata": {"kb_id": kb_id, "title": title, "chunk_index": idx, "word_count": len(words[start:end])}
        })
        idx += 1
        if end >= total:
            break
        start += 1800
    return records

def clean_title_from_filename(filename: str) -> str:
    """Extract and normalize document title from the PDF filename."""
    stem = Path(filename).stem
    stem = re.sub(r"^(tg_|ch_|doc_|\d+[\._\-\s]+)", "", stem, flags=re.IGNORECASE)
    stem = re.sub(r"\s*[\(\[\{].*?(?:pdfdrive|z-lib|libgen|download|ebook|scan|watermark).*?[\)\]\}]", "", stem, flags=re.IGNORECASE)
    stem = stem.replace("_", " ").replace("-", " ")
    stem = re.sub(r"\s+", " ", stem).strip(" .-_")
    return stem

def extract_clean_title(clean_md: str, filename: str) -> str:
    """Extract genuine document title, filtering out image tags, comments, and OCR artifacts."""
    fallback = clean_title_from_filename(filename)
    if not fallback or len(re.findall(r"[a-zA-Z0-9]", fallback)) < 2:
        fallback = "Document"

    lines = [line.strip() for line in clean_md.splitlines() if line.strip()]
    invalid_patterns = [
        r"^<!\-\-", r"^<[^>]+>$", r"^!\[", r"^\[\s*\]", r"^[-=*_#\s|]+$",
        r"^(table of contents|contents|index|chapter|preface|copyright|acknowledgments|license|edition)",
        r"^(image|figure|table|chart|diagram|photo|pic|picture|untitled|cover|page\s*\d+)",
        r"^(https?://|www\.|doi:|arxiv:)"
    ]

    detected_title = None
    for line in lines[:40]:
        text = re.sub(r"^#{1,6}\s*", "", line).strip()
        text = re.sub(r"<!--.*?-->", "", text).strip()
        text = re.sub(r"<[^>]+>", "", text).strip()
        text = re.sub(r"[*_~`]", "", text).strip()
        if not text:
            continue
        lower = text.lower()
        if any(re.search(p, lower) for p in invalid_patterns):
            continue
        letters = re.findall(r"[a-zA-Z]", text)
        if len(letters) < 4:
            continue
        if len(text.split()) > 20 or text.endswith((".", ";", ":")):
            continue
        if 5 <= len(text) <= 120:
            detected_title = text
            break

    is_generic_fn = any(fallback.lower() == g for g in ("document", "paper", "book", "file", "download", "untitled"))
    if detected_title and is_generic_fn:
        return detected_title
    if len(fallback) >= 6 and not is_generic_fn:
        return fallback
    return detected_title or fallback

def get_safe_filename(title: str, max_len: int = 120) -> str:
    """Generate a clean, filesystem-safe and GitHub-safe filename from document title."""
    s = re.sub(r"<!--.*?-->", "", title)
    s = re.sub(r"<[^>]+>", "", s)
    s = re.sub(r'[:/\\?*"<>|!#`]', " ", s)
    s = re.sub(r"\s+", " ", s).strip(" .-_#`!~")
    if not s or len(re.findall(r"[a-zA-Z0-9]", s)) < 2:
        s = "Document"
    if len(s) > max_len:
        s = s[:max_len].rsplit(" ", 1)[0].strip()
    return s

def push_github(repo, path: str, msg: str, content: str) -> None:
    """Push new file or update existing file in GitHub repository with support for >1MB files."""
    for attempt in range(3):
        try:
            sha = None
            try:
                item = repo.get_contents(path)
                sha = item.sha if not isinstance(item, list) else None
            except GithubException as ge:
                if ge.status == 404:
                    sha = None
                elif ge.status == 403:
                    parent = str(Path(path).parent).replace("\\", "/")
                    fname = Path(path).name
                    dir_path = "" if parent in (".", "") else parent
                    dir_items = repo.get_contents(dir_path)
                    if isinstance(dir_items, list):
                        for it in dir_items:
                            if it.name == fname:
                                sha = it.sha
                                break

            if sha:
                repo.update_file(path, msg, content, sha)
            else:
                try:
                    repo.create_file(path, msg, content)
                except GithubException as ce:
                    parent = str(Path(path).parent).replace("\\", "/")
                    fname = Path(path).name
                    dir_path = "" if parent in (".", "") else parent
                    dir_items = repo.get_contents(dir_path)
                    if isinstance(dir_items, list):
                        for it in dir_items:
                            if it.name == fname:
                                repo.update_file(path, msg, content, it.sha)
                                return
                    raise ce
            return
        except Exception as e:
            if attempt == 2:
                log.error("GitHub push failed permanently for %s: %s", path, e)
                raise
            time.sleep(2.0)

def commit_vault(kb_id: str, title: str, clean_md: str, dataset: str, reg: dict[str, list[str]]) -> tuple[bool, str]:
    if not GITHUB_TOKEN or not GITHUB_REPO_URL:
        log.error("GitHub sync skipped: GITHUB_TOKEN or GITHUB_REPO_URL is missing.")
        return False, "Missing credentials"
    try:
        gh = Github(auth=Auth.Token(GITHUB_TOKEN))
        repo = gh.get_repo(extract_repo_name(GITHUB_REPO_URL))
        safe_title = get_safe_filename(title)
        push_github(repo, f"text_vault/{safe_title}.md", f"feat: {kb_id} {safe_title[:50]}", clean_md)
        push_github(repo, f"dataset_vault/{safe_title}_dataset.jsonl", f"feat: {kb_id} dataset", dataset)
        push_github(repo, "registry.json", f"chore: register {kb_id}", json.dumps(reg, indent=2, ensure_ascii=False))
        readme_content = generate_master_readme(reg, GITHUB_REPO_URL)
        push_github(repo, "README.md", f"docs: update catalog for {kb_id}", readme_content)
        log.info("Successfully pushed %s (%s) and README to GitHub Vault!", kb_id, safe_title)
        return True, "OK"
    except Exception as e:
        log.exception("GitHub commit failed for %s: %s", kb_id, e)
        return False, str(e)

                if job.control_msg_id:
                    try:
                        await client.edit_message(TELEGRAM_GROUP_ID, job.control_msg_id, card, buttons=abort_btn, parse_mode="html")
                    except Exception:
                        pass

        await client.download_media(job.media_obj, file=str(temp_pdf), progress_callback=dl_progress)
        if job.cancelled:
            return

        # Step 2: Deduplication Check
        pdf_hash = compute_sha256(temp_pdf)
        raw_title = clean_title_from_filename(job.file_name)
        is_dup, dup_id, dup_reason = check_deduplication(pdf_hash, raw_title, state.registry)
        if is_dup:
            if job.control_msg_id:
                await client.edit_message(TELEGRAM_GROUP_ID, job.control_msg_id, f"⚠️ <b>[DUPLICATE SKIPPED]</b>\n📄 <code>{html.escape(job.file_name)}</code>\n🛡️ Matches <code>{dup_id}</code> ({html.escape(dup_reason)})\n<i>Skipped to save GPU time and prevent duplicates.</i>", parse_mode="html")
            return

        # Step 3: Docling Extraction
        state.highest_kb_num += 1
        kb_id = f"KB-{state.highest_kb_num:04d}"
        if job.control_msg_id:
            await client.edit_message(TELEGRAM_GROUP_ID, job.control_msg_id, f"🔄 <b>[INGESTING {kb_id}]</b>\n📥 Download: ✅\n⚙️ Docling GPU OCR & Extraction: ⏳ Running...", buttons=abort_btn, parse_mode="html")

        loop = asyncio.get_running_loop()
        clean_md, pages = await loop.run_in_executor(None, extract_pdf, temp_pdf)
        title = extract_clean_title(clean_md, job.file_name)
        words = len(clean_md.split())
        chunks = chunk_text(clean_md, kb_id, title)
        dataset_content = "\n".join([json.dumps(c, ensure_ascii=False) for c in chunks]) + "\n"

        # Local files named with actual title
        safe_title = get_safe_filename(title)
        temp_md = TEMP_DIR / f"{safe_title}.md"
        temp_md.write_text(f"---\nkb_id: \"{kb_id}\"\ntitle: \"{title}\"\npages: {pages}\nwords: {words}\n---\n\n{clean_md}", encoding="utf-8")
        temp_dset = TEMP_DIR / f"{safe_title}_dataset.jsonl"
        temp_dset.write_text(dataset_content, encoding="utf-8")
        temp_pdf_named = TEMP_DIR / f"{safe_title}.pdf"
        shutil.copy2(temp_pdf, temp_pdf_named)

        # Step 4: Deliver to Topics with actual names
        await send_topic(RAW_PDF_TOPIC_ID, f"📄 <b>[PDF] {html.escape(title)}</b>\n\n🆔 <code>{kb_id}</code>\n📊 {job.file_size_bytes/(1024*1024):.2f} MB | {pages} pages", file_path=temp_pdf_named)
        await send_topic(TEXT_MD_TOPIC_ID, f"📝 <b>[TEXT] {html.escape(title)}</b>\n\n🆔 <code>{kb_id}</code>\n📊 {words:,} words | {pages} pages\n✨ Zero-link sanitized", file_path=temp_md)
        await send_topic(DATASET_TOPIC_ID, f"📊 <b>[DATASET] {html.escape(title)}</b>\n\n🆔 <code>{kb_id}</code>\n🧩 {len(chunks)} Chunks | Unsloth & LLaMA-Factory ready", file_path=temp_dset)

        # Step 5: Update Registry & GitHub
        state.registry["papers"].append(f"tg_{job.file_name}")
        state.registry["hashes"].append(pdf_hash)
        state.registry["titles"].append(title)
        save_local_registry(state.registry)
        gh_ok, gh_err = await loop.run_in_executor(None, commit_vault, kb_id, title, clean_md, dataset_content, state.registry)

        # Mirror to Registry Topic 649 & General 1
        await send_topic(REGISTRY_TOPIC_ID, f"📚 <b>[REGISTRY UPDATE — {kb_id}]</b>\n<b>Title:</b> {html.escape(title)}\n📊 Vault Total: <code>{len(state.registry['papers'])}</code>\n🔒 SHA256: <code>{pdf_hash}</code>\n🐙 GitHub: {'✅ Pushed to Vault' if gh_ok else f'⚠️ Push Failed ({html.escape(gh_err[:40])})'}")
        await send_topic(GENERAL_TOPIC_ID, f"📢 <b>[VAULT INGESTION]</b>\n✅ Archived: <code>{kb_id}</code> — <i>{html.escape(title[:70])}</i>\n📚 Total Documents: <code>{len(state.registry['papers'])}</code>\n🐙 GitHub: {'✅ Synced' if gh_ok else 'Local only'}")

        # Completion Card in Topic 648
        dur = round(time.time() - start_t, 1)
        final_msg = f"✅ <b>[INGESTION COMPLETE — {kb_id}]</b>\n\n📄 <b>{html.escape(title)}</b>\n⏱️ Time: {dur}s\n📊 {words:,} words | {pages} pages | {len(chunks)} chunks\n\n📦 <b>Delivered To:</b>\n  • 📄 PDF -> Topic #354\n  • 📝 TEXT -> Topic #355\n  • 📊 DATASET -> Topic #356\n  • 📚 REGISTRY -> Topic #649\n  • 🐙 GITHUB -> {'✅ Pushed to Vault' if gh_ok else f'❌ Push Failed ({html.escape(gh_err[:40])})'}"
        if job.control_msg_id:
            await client.edit_message(TELEGRAM_GROUP_ID, job.control_msg_id, final_msg, parse_mode="html")
    except Exception as e:
        log.exception("Error: %s", e)
        if job.control_msg_id:
            await client.edit_message(TELEGRAM_GROUP_ID, job.control_msg_id, f"❌ <b>Ingestion Error:</b> <code>{html.escape(str(e))}</code>", parse_mode="html")
    finally:
        # Zero Disk Accumulation: delete raw PDF immediately
        if temp_pdf.exists():
            temp_pdf.unlink(missing_ok=True)
        if 'temp_pdf_named' in locals() and temp_pdf_named.exists():
            temp_pdf_named.unlink(missing_ok=True)
        if 'temp_md' in locals() and temp_md.exists():
            temp_md.unlink(missing_ok=True)
        if 'temp_dset' in locals() and temp_dset.exists():
            temp_dset.unlink(missing_ok=True)
        state.active_job = None

async def worker_loop():
    log.info("Queue Worker Loop started. Waiting for ingestion tasks...")
    while True:
        try:
            if not state.is_paused:
                job = await state.queue.get()
                await process_job(job)
                state.queue.task_done()
            await asyncio.sleep(1.0)
        except asyncio.CancelledError:
            log.info("Worker loop cancelled gracefully.")
            break
        except Exception as e:
            log.error("Worker loop exception: %s", e)
            await asyncio.sleep(2.0)

async def scan_channel_backlog():
    """Continuously monitors and scans @mybooksaspdf starting from ID 1 for PDFs, enqueuing them 1-by-1."""
    await asyncio.sleep(3.0)
    log.info("📡 Initializing backlog scanner for default channel @%s...", SOURCE_CHANNEL_USERNAME)
    try:
        channel = await client.get_entity(SOURCE_CHANNEL_ID)
    except Exception:
        try:
            channel = await client.get_entity(SOURCE_CHANNEL_USERNAME)
        except Exception as e:
            log.error("Could not resolve channel @%s: %s", SOURCE_CHANNEL_USERNAME, e)
            return

    curr_id = 1
    batch_size = 50
    empty_streak = 0

    while True:
        try:
            ids = list(range(curr_id, curr_id + batch_size))
            msgs = await client.get_messages(channel, ids=ids)
            valid = [m for m in msgs if m]

            if not valid:
                empty_streak += 1
                if empty_streak >= 3:
                    await asyncio.sleep(20)
                    empty_streak = 0
                    continue
                else:
                    curr_id += batch_size
                    await asyncio.sleep(0.5)
                    continue

            empty_streak = 0
            for m in valid:
                if m.id in state.processed_msg_ids:
                    continue
                if not m.file or not m.file.name or not m.file.name.lower().endswith(".pdf"):
                    continue

                state.processed_msg_ids.add(m.id)
                fname = m.file.name
                fsize = m.file.size or 0
                raw_title = clean_title_from_filename(fname)

                # Deduplication pre-check against current registry
                is_dup, dup_id, dup_reason = check_deduplication(None, raw_title, state.registry)
                if is_dup:
                    log.info("Channel PDF skipped (matches %s): %s", dup_id, fname)
                    continue

                eta = estimate_eta(fsize, "auto")
                job_id = f"chan_{m.id}_{int(time.time())}"
                job = IngestionJob(
                    job_id=job_id,
                    message_id=m.id,
                    topic_id=UPLOAD_TOPIC_ID,
                    chat_id=m.chat_id,
                    file_name=fname,
                    file_size_bytes=fsize,
                    media_obj=m.media,
                    mode="auto",
                    eta_seconds=eta,
                )
                card = (
                    f"📡 <b>[FOUND IN @{SOURCE_CHANNEL_USERNAME}]</b>\n\n"
                    f"📄 <b>File:</b> <code>{html.escape(fname)}</code>\n"
                    f"📊 <b>Size:</b> {fsize/(1024*1024):.2f} MB | ⏱️ <b>ETA:</b> ~{eta}s\n"
                    f"⏳ <b>Queue Position:</b> {state.queue.qsize() + 1}\n\n"
                    f"<i>Queued for 1-by-1 CUDA processing. Original remains safe.</i>"
                )
                ctrl = await send_topic(UPLOAD_TOPIC_ID, card, buttons=[[Button.inline("⏹️ Abort", data=f"abort_{job_id}".encode())]])
                if ctrl:
                    job.control_msg_id = ctrl.id
                await state.queue.put(job)
                log.info("Enqueued channel PDF #%d: %s", m.id, fname)

            curr_id += batch_size
            await asyncio.sleep(0.5)

        except asyncio.CancelledError:
            log.info("Channel backlog scanner cancelled gracefully.")
            break
        except Exception as e:
            log.warning("Channel scanner loop warning: %s", e)
            await asyncio.sleep(10)

@client.on(events.NewMessage)
async def on_message(event):
    msg = event.message
    chat_id = event.chat_id
    chat = await event.get_chat()
    chat_username = (getattr(chat, "username", "") or "").lower()
    is_source_channel = (
        chat_username == SOURCE_CHANNEL_USERNAME.lower()
        or chat_id == SOURCE_CHANNEL_ID
        or str(chat_id).endswith("3932114350")
    )
    is_target_group = (chat_id == TELEGRAM_GROUP_ID)
    is_dm = event.is_private

    if not is_source_channel and not is_target_group and not is_dm:
        return

    t_id = get_topic(msg)
    txt = (msg.text or "").strip()

    if is_target_group or is_dm:
        if txt.startswith("/status"):
            await event.reply(f"📊 <b>[STATUS]</b>\n📡 Channel: <code>@{SOURCE_CHANNEL_USERNAME}</code> (Read-Only)\n📚 Registered: <code>{len(state.registry['papers'])}</code>\n🆔 Next ID: <code>KB-{state.highest_kb_num + 1:04d}</code>\n⚡ Queue: {'⏸️ Paused' if state.is_paused else '🟢 Active'} ({state.queue.qsize()} waiting)\n🎛️ Mode: <code>{state.current_mode.upper()}</code>", parse_mode="html")
            return
        elif txt.startswith("/pause"):
            state.is_paused = True
            await event.reply("⏸️ Queue paused.", parse_mode="html")
            return
        elif txt.startswith("/resume"):
            state.is_paused = False
            await event.reply("▶️ Queue resumed.", parse_mode="html")
            return
        elif txt.startswith("/mode"):
            state.current_mode = "auto" if state.current_mode == "interactive" else "interactive"
            await event.reply(f"🎛️ Mode switched to <b>{state.current_mode.upper()}</b>", parse_mode="html")
            return
        elif txt.startswith("/sync"):
            state.registry, state.highest_kb_num = sync_github_registry(GITHUB_TOKEN, GITHUB_REPO_URL, state.registry)
            await event.reply(f"✅ Registry synced ({len(state.registry['papers'])} total)!", parse_mode="html")
            return

    # PDF Ingestion from @mybooksaspdf or Topic 648
    if is_target_group and t_id != UPLOAD_TOPIC_ID:
        return
    doc = msg.document
    if not doc:
        return

    fname = "document.pdf"
    for a in getattr(doc, "attributes", []):
        if hasattr(a, "file_name") and a.file_name:
            fname = a.file_name
            break

    if not fname.lower().endswith(".pdf"):
        return

    fsize = getattr(doc, "size", 0)
    job_id = f"job_{int(time.time()*1000)}"
    eta = estimate_eta(fsize, "deep")

    # Duplicate pre-check
    raw_title = clean_title_from_filename(fname)
    is_dup, dup_id, dup_reason = check_deduplication(None, raw_title, state.registry)
    if is_dup:
        log.info("Duplicate skipped: %s matches %s", fname, dup_id)
        if not is_source_channel:
            await msg.reply(f"⚠️ <b>[DUPLICATE DETECTED]</b>\n<code>{html.escape(fname)}</code> matches <code>{dup_id}</code>.\nSkipped to prevent duplicate storage.", parse_mode="html")
        return

    # Job created (STRICTLY READ-ONLY from channel: NEVER deletes from channel)
    job = IngestionJob(job_id=job_id, message_id=msg.id, topic_id=UPLOAD_TOPIC_ID, chat_id=event.chat_id, file_name=fname, file_size_bytes=fsize, media_obj=msg.media, eta_seconds=eta)

    if is_source_channel:
        if msg.id in state.processed_msg_ids:
            return
        state.processed_msg_ids.add(msg.id)
        chan_card = (
            f"📡 <b>[NEW BOOK IN @{SOURCE_CHANNEL_USERNAME}]</b>\n\n"
            f"📄 <b>File:</b> <code>{html.escape(fname)}</code>\n"
            f"📊 <b>Size:</b> {fsize/(1024*1024):.2f} MB | ⏱️ <b>ETA:</b> ~{eta}s\n"
            f"⏳ <b>Queue Position:</b> {state.queue.qsize() + 1}\n\n"
            f"<i>Original PDF remains safe & untouched in channel.</i>"
        )
        ctrl = await send_topic(UPLOAD_TOPIC_ID, chan_card, buttons=[[Button.inline("⏹️ Abort", data=f"abort_{job_id}".encode())]])
        if ctrl:
            job.control_msg_id = ctrl.id
        await state.queue.put(job)
        return

    if state.current_mode == "auto":
        ctrl = await msg.reply(f"📄 <b>[QUEUED — AUTO INGEST]</b>\n<b>File:</b> <code>{html.escape(fname)}</code> ({fsize/(1024*1024):.2f} MB)\n⏱️ ETA: ~{eta}s", buttons=[[Button.inline("⏹️ Abort", data=f"abort_{job_id}".encode())]], parse_mode="html")
        job.control_msg_id = ctrl.id
        await state.queue.put(job)
    else:
        state.pending_interactive[job_id] = job
        ctrl = await msg.reply(f"📄 <b>[DOCUMENT RECEIVED — USER CONTROL]</b>\n<b>File:</b> <code>{html.escape(fname)}</code> ({fsize/(1024*1024):.2f} MB)\n⏱️ ETA: ~{eta}s\n\nSelect processing mode:", buttons=[
            [Button.inline("▶️ Ingest (Deep SFT)", data=f"start_deep_{job_id}".encode()), Button.inline("⚡ Fast (Chunks)", data=f"start_fast_{job_id}".encode())],
            [Button.inline("❌ Cancel", data=f"cancel_{job_id}".encode())]
        ], parse_mode="html")
        job.control_msg_id = ctrl.id

@client.on(events.CallbackQuery)
async def on_callback(event):
    data = event.data.decode("utf-8")
    if data.startswith("start_"):
        _, mode, j_id = data.split("_", 2)
        job = state.pending_interactive.pop(j_id, None)
        if job:
            job.mode = mode
            await event.edit(f"⏳ <b>[QUEUED]</b> <code>{html.escape(job.file_name)}</code>\nMode: {mode.upper()} | ETA: ~{job.eta_seconds}s", buttons=[[Button.inline("⏹️ Abort", data=f"abort_{job.job_id}".encode())]], parse_mode="html")
            await state.queue.put(job)
    elif data.startswith("cancel_"):
        state.pending_interactive.pop(data.removeprefix("cancel_"), None)
        await event.edit("❌ Ingestion cancelled.", parse_mode="html")
    elif data.startswith("abort_"):
        j_id = data.removeprefix("abort_")
        if state.active_job and state.active_job.job_id == j_id:
            state.active_job.cancelled = True
        await event.edit("⏹️ Aborted.", parse_mode="html")

async def main():
    state.registry = load_local_registry()
    await client.start(bot_token=TELEGRAM_BOT_TOKEN)
    state.registry, state.highest_kb_num = sync_github_registry(GITHUB_TOKEN, GITHUB_REPO_URL, state.registry)
    gh_status = "Connected" if GITHUB_TOKEN else "Token Missing"
    log.info("Knowledge Bot Ready! %d papers in registry. GitHub: %s", len(state.registry['papers']), gh_status)
    
    # Store strong references so Python GC never destroys pending tasks
    w_task = create_monitored_task(worker_loop())
    s_task = create_monitored_task(scan_channel_backlog())
    
    await send_topic(GENERAL_TOPIC_ID, f"🤖 <b>[KNOWLEDGE BOT ONLINE — T4 GPU READY]</b>\n📚 Total Documents: <code>{len(state.registry['papers'])}</code>\n📡 Monitored Channel: <code>@{SOURCE_CHANNEL_USERNAME}</code> (Continuous Backlog & Real-time Ingestion)\nDrop PDFs in Topic #648 for manual ingestion!")
    await send_topic(UPLOAD_TOPIC_ID, f"🚀 <b>[USER CONTROL CENTER READY]</b>\n📡 Ingesting PDFs from <code>@{SOURCE_CHANNEL_USERNAME}</code> 1-by-1.\nDrop any PDF here (up to 2GB) to convert and archive!")
    
    try:
        await client.run_until_disconnected()
    finally:
        w_task.cancel()
        s_task.cancel()
        await asyncio.gather(w_task, s_task, return_exceptions=True)

if __name__ == "__main__":
    try:
        import nest_asyncio
        nest_asyncio.apply()

        pass
    try:
        asyncio.run(main())
    except RuntimeError as re:

            loop = asyncio.get_event_loop()
            create_monitored_task(main())
        else:
            raise
    except (KeyboardInterrupt, SystemExit):
        pass
